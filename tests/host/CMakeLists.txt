cmake_minimum_required(VERSION 3.20)
project(ro_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()

function(add_ro_host_test target source)
    add_executable(${target} ${source} ../../components/ro_core/src/controller.cpp)
    target_include_directories(${target} PRIVATE ../../components/ro_core/include)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME ${target} COMMAND ${target})
endfunction()

add_ro_host_test(ro_host_tests test_core.cpp)
add_ro_host_test(ro_regression_tests test_regressions.cpp)
add_ro_host_test(ro_plan_regression_tests test_plan_regressions.cpp)
