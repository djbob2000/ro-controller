cmake_minimum_required(VERSION 3.20)
project(ro_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(ro_host_tests test_core.cpp ../../components/ro_core/src/controller.cpp)
target_include_directories(ro_host_tests PRIVATE ../../components/ro_core/include)
target_compile_options(ro_host_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME ro_host_tests COMMAND ro_host_tests)
